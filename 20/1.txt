data() {
    return {
        hasNewInquiries: false // Флаг наличия новых запросов
    };
},
mounted() {
    this.checkNewInquiries();
    // Опционально: опрос сервера каждые 30 секунд для актуальности
    this.timer = setInterval(this.checkNewInquiries, 30000);
},
beforeDestroy() {
    clearInterval(this.timer);
},
methods: {
    async checkNewInquiries() {
        try {
            const response = await fetch('/af/inquiries/check-new', {
                headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` }
            });
            const data = await response.json();
            this.hasNewInquiries = data.has_new; // true или false
        } catch (e) {
            console.error("Ошибка проверки новых запросов:", e);
        }
    }
}

    <span v-if="hasNewInquiries" class="menu-badge-dot" title="Есть новые запросы!"></span>
</router-link>
