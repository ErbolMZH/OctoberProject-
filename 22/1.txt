	"fmt"
	"net/http"

	"your_project/comps"

// RequestDTO для принятия ID от фронтенда
type UpdateSubjectReq struct {
	ID string `json:"id"`
}

// UpdateSubjectByIDHandler — обработчик запроса обновления субъекта
func UpdateSubjectByIDHandler(c echo.Context) error {
	var req UpdateSubjectReq
	if err := c.Bind(&req); err != nil || req.ID == "" {
		return c.JSON(http.StatusBadRequest, echo.Map{
			"error": "Укажите корректный ID субъекта",
		})
	}

	// 1. Проверка авторизации / получение токенов (как в main)[cite: 2]
	authData := model.AFGetAuthStruct()
	secAfterTokenCreation := int(time.Since(authData.CreatedAt).Seconds())
	if secAfterTokenCreation >= authData.RefreshExpiresIn {
		var err error
		authData, _, err = model.AFAuth("faces token expired")
		if err != nil {
			logger.Log("ERROR", "target_subject.go", "NULL", fmt.Sprintf("AFAuth fiasco: %s", err), err)
			return c.JSON(http.StatusInternalServerError, echo.Map{
				"error": "Ошибка авторизации во внешнем сервисе",
			})
		}
		model.AFGlobalAuthData = authData
	}

	// 2. Логирование и формирование URL для внешнего запроса[cite: 6]
	vPerson := req.ID
	logger.Log("Info", "target_subject.go", "NULL", fmt.Sprintf("AFRequestPlain : %s", vPerson), nil)

	url := comps.Config.AF.APIURL + "/external/v2/lists/subject/" + vPerson + "AFPersonInfo"
	body1, status, err := model.AFRequestPlain("GET", url, authData.AccessToken)

	if err != nil {
		logger.Log("ERROR", "target_subject.go", "NULL", fmt.Sprintf("AFRequestPlain fiasco: %s", err), err)
		return c.JSON(http.StatusInternalServerError, echo.Map{
			"error": "Ошибка при получении данных субъекта с внешнего API",
		})
	}

	if status == 401 || status == 403 || status == 404 {
		logger.Log("ERROR", "target_subject.go", "NULL", fmt.Sprintf("AFRequestPlain returned %d", status), nil)
		return c.JSON(status, echo.Map{
			"error": fmt.Sprintf("Внешний сервис вернул статус: %d", status),
		})
	}

	// 3. Сохранение данных в PostgreSQL и отправка в ИБСО (согласно вашему коду)[cite: 1, 3]
	if len(body1) > 0 {
		// Отправка структуры в ИБСО при необходимости[cite: 3]
		bodyOra := fmt.Sprintf(`{"total":1,"limit":1,"offset":0,"items":[ %s ]}`, body1)
		errOra := model.OraSendToIBSO("F_SETCLARR", bodyOra)
		if errOra != nil {
			logger.Log("ERROR", "target_subject.go", "NULL", fmt.Sprintf("OraSendToIBSO fiasco: %s", errOra), errOra)
		}

		// Сохранение JSON субъекта в PostgreSQL[cite: 1, 3]
		errDB := AF_JSON_SUBJECT(string(body1))
		if errDB != nil {
			logger.Log("ERROR", "target_subject.go", "NULL", fmt.Sprintf("AF_JSON_SUBJECT fiasco: %s", errDB), errDB)
			return c.JSON(http.StatusInternalServerError, echo.Map{
				"error": "Ошибка при сохранении субъекта в базу данных",
			})
		}

		logger.Log("INFO", "target_subject.go", "NULL", fmt.Sprintf("AF_JSON_SUBJECT success for %s, len: %d", vPerson, len(body1)), nil)
	} else {
		logger.Log("INFO", "target_subject.go", "NULL", fmt.Sprintf("Empty body returned for %s", vPerson), nil)
	}

	// 4. Успешный ответ
	return c.JSON(http.StatusOK, echo.Map{
		"success": true,
		"message": "Данные субъекта успешно обновлены",
		"id":      vPerson,
	})
}

        :disabled="isUpdating" 
        @click="updateSubjectData"
      >
        <span v-if="isUpdating">Обновление...</span>
        <span v-else>Получить новые обновления по ID</span>
      </button>

  props: {

    };

      this.isUpdating = true;
      try {
        const response = await axios.post('/api/v1/subject/update', {
          id: this.subjectId
        });

        if (response.data.success) {
          // Уведомление об успехе
          this.$toast ? this.$toast.success('Данные успешно обновлены!') : alert('Данные успешно обновлены!');
          
          // Перезапрашиваем актуальные данные из БД для обновления таблицы/экрана
          this.fetchSubjectDetails();
        }
      } catch (error) {
        console.error('Ошибка при обновлении субъекта:', error);
        const errMsg = error.response?.data?.error || 'Произошла ошибка при запросе обновлений';
        this.$toast ? this.$toast.error(errMsg) : alert(errMsg);
      } finally {
        this.isUpdating = false;
      }
    },
    async fetchSubjectDetails() {
      // Ваш существующий метод загрузки данных субъекта из БД
    }
  }
};
</script>
