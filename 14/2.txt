
// 1. Функция для приведения даты из Go ("2026-10-19T00:00:00Z") к формату JS ("19.10.2026")
const formatGoDate = (goDateStr) => {
    if (!goDateStr) return '';
    const datePart = goDateStr.split('T')[0]; // получаем "2026-10-19"
    const [year, month, day] = datePart.split('-');
    return `${day}.${month}.${year}`; // собираем "19.10.2026"
};

// 2. Функция для очистки JS-даты ("28.09.2026 00:00:00" -> "28.09.2026")
const formatJsDate = (jsDateStr) => {
    if (!jsDateStr) return '';
    return jsDateStr.split(' ')[0];
};
